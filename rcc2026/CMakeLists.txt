cmake_minimum_required(VERSION 3.20)
project(RccCompilerResearch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(RCC_LUAU_SOURCE_DIR "" CACHE PATH "Clean upstream Luau 0.743 checkout")
if(NOT EXISTS "${RCC_LUAU_SOURCE_DIR}/Compiler/src/Compiler.cpp")
    message(FATAL_ERROR "Set RCC_LUAU_SOURCE_DIR to an upstream Luau 0.743 checkout")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${RCC_LUAU_SOURCE_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE RCC_LUAU_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT RCC_LUAU_COMMIT STREQUAL "9d7748a4f324213a18cb8283fa2a77c330225fa0")
    message(FATAL_ERROR "Luau checkout does not match the examined 0.743 source revision")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${RCC_LUAU_SOURCE_DIR}" status --porcelain --untracked-files=no
    OUTPUT_VARIABLE RCC_LUAU_CHANGES OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT RCC_LUAU_CHANGES STREQUAL "")
    message(FATAL_ERROR "Luau tracked source files are modified; use a clean checkout")
endif()
set(LUAU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_WEB OFF CACHE BOOL "" FORCE)
add_subdirectory("${RCC_LUAU_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/luau")
add_executable(rcc-compiler-smoke luau_roundtrip.cpp)
target_link_libraries(rcc-compiler-smoke PRIVATE Luau.Compiler Luau.VM)
enable_testing()
add_test(NAME compiler_vm_roundtrip COMMAND rcc-compiler-smoke)
